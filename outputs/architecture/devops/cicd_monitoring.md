## 1. CI/CD Pipeline (GitHub Actions)

### Branch → Environment Mapping

| Branch | Environment | Trigger | Deploy Method |
|--------|------------|---------|---------------|
| `feature/*` | None | Push | Run lint + tests only |
| `staging` | Staging | Merge (auto) | Rolling ECS update |
| `main` | Production | Manual trigger | Blue-green ECS swap |
| `hotfix/*` | Production | Manual trigger | Rolling ECS update |

---

### Pipeline Definition

```yaml
# .github/workflows/ci.yml — runs on all branches
name: CI

on: [push, pull_request]

jobs:
  lint-and-typecheck:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: '22' }
      - run: npm ci
      - run: npm run typecheck      # tsc --noEmit
      - run: npm run lint           # ESLint zero-tolerance

  security-scan:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - run: npm audit --audit-level=critical   # blocks on critical CVEs
      - uses: gitleaks/gitleaks-action@v2       # blocks on detected secrets

  test:
    runs-on: ubuntu-latest
    services:
      postgres:
        image: postgres:16
        env: { POSTGRES_DB: test_db, POSTGRES_PASSWORD: test }
      redis:
        image: redis:7
    steps:
      - uses: actions/checkout@v4
      - run: npm ci
      - run: npm run test:unit        # Vitest unit + integration tests
      - run: npm run test:coverage    # Fail if coverage < 80%

  e2e:
    runs-on: ubuntu-latest
    if: github.ref == 'refs/heads/staging' || github.base_ref == 'main'
    steps:
      - uses: actions/checkout@v4
      - run: npm ci
      - run: npx playwright install --with-deps chromium
      - run: npm run test:e2e         # Full candidate journey
      - run: npm run test:a11y        # axe-core WCAG scan

  build:
    needs: [lint-and-typecheck, security-scan, test]
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: aws-actions/configure-aws-credentials@v4
        with:
          role-to-assume: ${{ secrets.AWS_CI_ROLE_ARN }}
          aws-region: us-east-1
      - uses: aws-actions/amazon-ecr-login@v2
      - run: |
          docker build -t $ECR_REGISTRY/interview-prep-api:$GITHUB_SHA .
          docker push $ECR_REGISTRY/interview-prep-api:$GITHUB_SHA
```

```yaml
# .github/workflows/deploy-staging.yml
name: Deploy Staging

on:
  push:
    branches: [staging]

jobs:
  deploy:
    runs-on: ubuntu-latest
    environment: staging
    steps:
      - uses: aws-actions/configure-aws-credentials@v4
      - name: Update ECS service
        run: |
          aws ecs update-service \
            --cluster interview-prep-staging \
            --service interview-prep-api \
            --force-new-deployment
      - name: Wait for stable
        run: |
          aws ecs wait services-stable \
            --cluster interview-prep-staging \
            --services interview-prep-api
      - name: Smoke test
        run: curl -f https://staging.api.platform.com/health
```

```yaml
# .github/workflows/deploy-production.yml
name: Deploy Production

on:
  workflow_dispatch:         # Manual trigger only
    inputs:
      image_tag:
        description: 'ECR image tag to deploy'
        required: true

jobs:
  deploy:
    runs-on: ubuntu-latest
    environment: production  # Requires GitHub environment approval
    steps:
      - uses: aws-actions/configure-aws-credentials@v4
      - name: Blue-green deploy
        run: |
          # Register new task definition with new image tag
          # Update CodeDeploy deployment group (blue-green ECS)
          # Wait for health checks to pass before routing traffic
          aws deploy create-deployment \
            --application-name interview-prep \
            --deployment-group-name interview-prep-prod \
            --revision imageFile=${{ inputs.image_tag }}
      - name: Verify production health
        run: curl -f https://api.platform.com/health
```

## 2. Docker Build Standards

```dockerfile
# Dockerfile (multi-stage — API service)
FROM node:22-alpine AS deps
WORKDIR /app
COPY package*.json ./
RUN npm ci --only=production

FROM node:22-alpine AS builder
WORKDIR /app
COPY --from=deps /app/node_modules ./node_modules
COPY . .
RUN npm run build

FROM node:22-alpine AS runner
WORKDIR /app
ENV NODE_ENV=production
RUN addgroup --system --gid 1001 nodejs && \
    adduser --system --uid 1001 fastify
COPY --from=builder --chown=fastify:nodejs /app/dist ./dist
COPY --from=deps --chown=fastify:nodejs /app/node_modules ./node_modules
USER fastify
EXPOSE 3001
HEALTHCHECK --interval=30s --timeout=5s --retries=3 \
  CMD wget -qO- http://localhost:3001/health || exit 1
CMD ["node", "dist/server.js"]
```

## 3. Monitoring Design

### 3.1 CloudWatch Alarms

```hcl
# terraform/modules/monitoring/alarms.tf

# API Error Rate > 1% over 5 minutes — P1 Alert
resource "aws_cloudwatch_metric_alarm" "api_error_rate" {
  alarm_name          = "api-error-rate-high"
  comparison_operator = "GreaterThanThreshold"
  evaluation_periods  = 2
  metric_name         = "5XXError"
  namespace           = "AWS/ApplicationELB"
  period              = 300
  statistic           = "Average"
  threshold           = 0.01
  alarm_actions       = [aws_sns_topic.p1_alerts.arn]
  ok_actions          = [aws_sns_topic.p1_alerts.arn]
}

# Score report p95 latency > 90s — P2 Alert
resource "aws_cloudwatch_metric_alarm" "score_report_latency" {
  alarm_name          = "score-report-latency-high"
  comparison_operator = "GreaterThanThreshold"
  evaluation_periods  = 3
  metric_name         = "score_report_latency_p95"
  namespace           = "InterviewPrep/Scoring"
  period              = 300
  extended_statistic  = "p95"
  threshold           = 90
  alarm_actions       = [aws_sns_topic.p2_alerts.arn]
}

# STT queue depth > 50 — P2 Alert
resource "aws_cloudwatch_metric_alarm" "stt_queue_depth" {
  alarm_name          = "stt-queue-depth-high"
  comparison_operator = "GreaterThanThreshold"
  evaluation_periods  = 1
  metric_name         = "stt_queue_depth"
  namespace           = "InterviewPrep/Queue"
  period              = 60
  statistic           = "Maximum"
  threshold           = 50
  alarm_actions       = [aws_sns_topic.p2_alerts.arn]
}

# Whisper instance unhealthy — P1 Alert
resource "aws_cloudwatch_metric_alarm" "whisper_health" {
  alarm_name          = "whisper-instance-unhealthy"
  comparison_operator = "LessThanThreshold"
  evaluation_periods  = 2
  metric_name         = "StatusCheckFailed"
  namespace           = "AWS/EC2"
  period              = 60
  statistic           = "Maximum"
  threshold           = 1
  alarm_actions       = [aws_sns_topic.p1_alerts.arn]
}

# Failed logins > 50/min — Security Alert
resource "aws_cloudwatch_metric_alarm" "failed_logins" {
  alarm_name          = "failed-login-rate-high"
  comparison_operator = "GreaterThanThreshold"
  evaluation_periods  = 1
  metric_name         = "failed_login_count"
  namespace           = "InterviewPrep/Auth"
  period              = 60
  statistic           = "Sum"
  threshold           = 50
  alarm_actions       = [aws_sns_topic.security_alerts.arn]
}
```

### 3.4 Alert Routing

| Severity | Channel | Response Time |
|----------|---------|--------------|
| P1 | PagerDuty (on-call rotation) + Slack #incidents | Immediate (24/7) |
| P2 | Slack #platform-alerts | Business hours; resolve within 4h |
| Security | Slack #security-alerts + Security Lead email | Within 1h |
| Info | Slack #platform-info | No action required |

### 3.5 Sentry Configuration

```typescript
// Sentry initialized in API server startup
Sentry.init({
  dsn: process.env.SENTRY_DSN,
  environment: process.env.NODE_ENV,
  tracesSampleRate: 0.1,        // 10% transaction tracing in production
  beforeSend(event) {
    // Strip PII from error events
    if (event.user) delete event.user.email
    if (event.request?.data) sanitizePII(event.request.data)
    return event
  },
})
```
